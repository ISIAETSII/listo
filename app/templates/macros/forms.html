{# Macros para pintar campos de WTForms con las clases de Bootstrap. Uso: {% from "macros/forms.html" import render_field %} ... {{ render_field(form.title) }} `kwargs` es una variable especial de Jinja: recoge los argumentos extra con los que se llama a la macro (por ejemplo, autofocus=true o rows=4) y se los pasamos tal cual a la etiqueta HTML. El operador `~` concatena cadenas. #} {% macro render_field(field) %}
{{ field.label(class="form-label") }} {{ field(class="form-control" ~ (" is-invalid" if field.errors else ""), **kwargs) }} {% for error in field.errors %}
{{ error }}
{% endfor %}
{% endmacro %} {% macro render_checkbox(field) %}
{{ field(class="form-check-input") }} {{ field.label(class="form-check-label") }}
{% endmacro %}
